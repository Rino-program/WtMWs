class TestDisplay:
    """Displayクラスのテスト"""
    
    def test_print_success(self, capsys):
        """成功メッセージのテスト"""
        Display.print_success("テスト成功")
        captured = capsys.readouterr()
        assert "テスト成功" in captured.out
        assert "✅" in captured.out
    
    def test_print_error(self, capsys):
        """エラーメッセージのテスト"""
        Display.print_error("エラーテスト")
        captured = capsys.readouterr()
        assert "エラーテスト" in captured.out
        assert "❌" in captured.out
    
    def test_print_warning(self, capsys):
        """警告メッセージのテスト"""
        Display.print_warning("警告テスト")
        captured = capsys.readouterr()
        assert "警告テスト" in captured.out
        assert "⚠️" in captured.out
    
    def test_print_status_bar(self, capsys):
        """ステータスバーのテスト"""
        Display.print_status_bar("テスト", 75, 100)
        captured = capsys.readouterr()
        assert "テスト" in captured.out
        assert "75/100" in captured.out
        assert "75.0%" in captured.out
    
    def test_print_table(self, capsys):
        """テーブル表示のテスト"""
        headers = ["列1", "列2"]
        rows = [["値1", "値2"], ["値3", "値4"]]
        Display.print_table(headers, rows)
        captured = capsys.readouterr()
        assert "列1" in captured.out
        assert "値1" in captured.out
    
    def test_confirm_yes(self, monkeypatch):
        """確認プロンプト（Yes）のテスト"""
        monkeypatch.setattr('builtins.input', lambda _: 'y')
        result = Display.confirm("テストですか？")
        assert result == True
    
    def test_confirm_no(self, monkeypatch):
        """確認プロンプト（No）のテスト"""
        monkeypatch.setattr('builtins.input', lambda _: 'n')
        result = Display.confirm("テストですか？")
        assert result == False


class TestAquariumCLI:
    """AquariumCLIクラスのテスト"""
    
    @pytest.fixture
    def cli(self):
        """CLIインスタンスのフィクスチャ"""
        return AquariumCLI()
    
    def test_initialization(self, cli):
        """初期化のテスト"""
        assert cli.aquarium is not None
        assert cli.game_engine is not None
        assert len(cli.aquarium.tanks) > 0
        assert cli.aquarium.money > 0

        cli.do_status("")
        captured = capsys.readouterr()
        assert "水族館ステータス" in captured.out
        assert "資金" in captured.out
        assert "評判" in captured.out

        cli.do_list("fish")
        captured = capsys.readouterr()
        assert "魚のリスト" in captured.out

        """list tanksコマンドのテスト"""
        cli.do_list("tanks")
        captured = capsys.readouterr()
        assert "水槽のリスト" in captured.out
    
    def test_do_feed_all(self, cli):
        """feedコマンド（全体）のテスト"""
        initial_money = cli.aquarium.money
        initial_hunger = cli.aquarium.tanks[0].fish_list[0].hunger

        initial_money = cli.aquarium.money
        cli.do_feed("1")
        assert cli.aquarium.money < initial_money

        # 水槽を汚す

        initial_money = cli.aquarium.money
        
        cli.do_clean("1")
        
        # 資金が減り、清潔度が上がる
        assert cli.aquarium.money < initial_money
        assert cli.aquarium.tanks[0].water_quality['cleanliness'] > 50.0
    
    def test_do_add_fish(self, cli):

        initial_money = cli.aquarium.money

class TestIntegration:
    """統合テスト"""
    
    def test_full_game_flow(self):
        """完全なゲームフローのテスト"""
        cli = AquariumCLI()
        
        # 初期状態確認
        assert cli.aquarium.money > 0
        assert len(cli.aquarium.tanks) > 0
        
        # 魚に餌をやる
        cli.do_feed("")
        
        # 水槽を掃除
        cli.do_clean("1")
        
        # 魚を追加
        cli.do_add_fish("goldfish テスト")
        
        # 1日進める
        cli.do_next_day("")
        
        # レポート確認
        cli.do_report("")
        
        # ゲームが正常に動作していることを確認
        assert cli.aquarium.current_day > 0
    
    def test_multiple_days_simulation(self):
        """複数日シミュレーションのテスト"""
        cli = AquariumCLI()
        initial_day = cli.aquarium.current_day
        
        # 10日進める
        for _ in range(10):
            cli.do_next_day("")
        
        assert cli.aquarium.current_day == initial_day + 10
    
    def test_money_management(self):
        """資金管理のテスト"""
        cli = AquariumCLI()
        
        # 初期資金を記録
        initial_money = cli.aquarium.money
        
        # いくつか操作を実行
        cli.do_feed("")  # 支出
        cli.do_clean("1")  # 支出
        cli.do_next_day("")  # 収入・支出
        
        # 資金が変動していることを確認
        assert cli.aquarium.money != initial_money

❌ 問題 1: coloramaの色が表示されない

症状： カラーコードがそのまま表示される

原因： Windowsターミナルでcoloramaが初期化されていない

解決方法：

# display.pyの先頭で必ず初期化
from colorama import init
init(autoreset=True)  # 自動リセットを有効化

# または、Windowsの場合は環境変数を設定
set PYTHONIOENCODING=utf-8

❌ 問題 2: コマンドが認識されない

症状： コマンドを入力しても「不明なコマンド」と表示される

原因： メソッド名がdo_コマンド名の形式になっていない

解決方法：

# ❌ 間違い
def command_status(self, arg):
    pass

# ✅ 正しい
def do_status(self, arg):
    pass

❌ 問題 3: テーブルが崩れる

症状： tabulateのテーブル表示が崩れる

原因： 日本語の文字幅の問題

解決方法：

# テーブルフォーマットを変更
Display.print_table(headers, rows, tablefmt="simple")

# または、文字幅を揃える
headers = ["ID      ", "名前    ", "状態    "]

❌ 問題 4: 入力でエラーが発生する

症状： 特定の入力でプログラムがクラッシュする

原因： 入力検証が不十分

解決方法：

# 必ず入力検証を行う
def do_add_fish(self, arg):
    # 引数チェック
    if not arg:
        Display.print_error("引数が必要です")
        return
    
    # 数値変換のエラーハンドリング
    try:
        value = int(arg)
    except ValueError:
        Display.print_error("無効な数値です")
        return
    
    # 範囲チェック
    if value < 0:
        Display.print_error("正の数を指定してください")
        return

❌ 問題 9: メモリ使用量が増え続ける

症状： 長時間プレイするとメモリ使用量が増加

原因： イベント履歴やログが蓄積

解決方法：

# 履歴の最大数を制限
class GameEngine:
    def __init__(self, aquarium):
        self.event_history = []
        self.max_history = 100  # 最大100件
    
    def add_event(self, event):
        self.event_history.append(event)
        # 古いイベントを削除
        if len(self.event_history) > self.max_history:
            self.event_history = self.event_history[-self.max_history:]

❌ 問題 10: Unicode文字が表示されない

症状： 絵文字や特殊文字が?になる

原因： ターミナルのエンコーディング設定

解決方法：

# Windowsの場合、コンソールをUTF-8に設定
chcp 65001

# Pythonスクリプトの先頭に追加
# -*- coding: utf-8 -*-
import sys
import io
sys.stdout = io.TextIOWrapper(sys.stdout.buffer, encoding='utf-8')
