最適化テクニック

1. リスト内包表記の使用


# ❌ 遅い
result = []
for i in range(1000):
    if i % 2 == 0:
        result.append(i * 2)

# ✅ 速い
result = [i * 2 for i in range(1000) if i % 2 == 0]

2. ジェネレータの活用


# ❌ メモリ大量消費
def get_all_fish_at_once():
    return [create_fish() for _ in range(1000000)]

# ✅ メモリ効率的
def get_fish_generator():
    for _ in range(1000000):
        yield create_fish()
