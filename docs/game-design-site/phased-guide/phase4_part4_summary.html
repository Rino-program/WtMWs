バランス調整の基本原則

  • 初心者向け魚（common）は安価で丈夫
  • レア度が上がるほど価格と難易度が上昇
  • メンテナンスコストは価格に比例
  • 魅力度（appeal）はレアリティと相関

import pytest
import json
from pathlib import Path
from utils.data_loader import DataLoader

def test_fish_species_json_valid():
    """fish_species.jsonが正しいJSON形式か"""
    with open('data/fish_species.json') as f:
        data = json.load(f)
    assert 'species' in data
    assert len(data['species']) == 15

def test_all_fish_have_required_fields():
    """すべての魚が必須フィールドを持つか"""
    loader = DataLoader()
    data = loader.load_fish_species()
    
    required_fields = ['id', 'name', 'category', 'rarity']
    for species in data['species']:
        for field in required_fields:
            assert field in species

def test_dataloader_caching():
    """DataLoaderのキャッシング動作"""
    loader = DataLoader()
    data1 = loader.load_fish_species()
    data2 = loader.load_fish_species()
    assert data1 is data2  # 同じオブジェクト

def test_create_fish_from_species():
    """魚オブジェクトの生成"""
    loader = DataLoader()
    fish = loader.create_fish_from_species("goldfish", "Test")
    assert fish.name == "Test"
    assert fish.species == "金魚"
