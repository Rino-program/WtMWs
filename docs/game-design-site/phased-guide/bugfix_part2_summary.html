1. Pythonロギング


import logging

# ロガーの設定
logging.basicConfig(
    level=logging.DEBUG,
    format='%(asctime)s - %(name)s - %(levelname)s - %(message)s',
    handlers=[
        logging.FileHandler('game.log'),
        logging.StreamHandler()
    ]
)

logger = logging.getLogger(__name__)

# 使用例
logger.debug("デバッグ情報")
logger.info("一般情報")
logger.warning("警告")
logger.error("エラー")

def problematic_function():
    x = 10
    pdb.set_trace()  # ブレークポイント
    y = x * 2
    return y

# s - step (関数内に入る)
# c - continue (次のブレークポイントまで)

3. プロファイリング


import cProfile
import pstats

# プロファイル実行
cProfile.run('main()', 'profile_stats')

# 結果分析
p = pstats.Stats('profile_stats')
p.sort_stats('cumulative')
p.print_stats(20)  # Top 20表示
