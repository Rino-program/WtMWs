"""

Pythonのcmdモジュールを使用したインタラクティブシェル

import cmd

class AquariumCLI(cmd.Cmd):
    """

    15個のコマンドを提供：
    - status: 全体の状態表示
    - list: 詳細リスト表示

    - sell: アイテム売却
    - next_day: 時間進行
    - skip: 複数日スキップ
    - save: セーブ
    - load: ロード
    - report: レポート生成
    - achievements: 実績表示
    - help: ヘルプ
    - quit: 終了

    intro = f"""{Colors.BRIGHT}{Colors.CYAN}
╔═══════════════════════════════════════════════════════════╗
║                                                           ║
║        🐠 バーチャル水族館ゲーム 🐠                        ║

║                                                           ║
╚═══════════════════════════════════════════════════════════╝
{Colors.RESET}

コマンドのヘルプを見るには 'help' または '?' を入力してください。

    def _initialize_game(self):

    def do_status(self, arg):
        """
        水族館の全体ステータスを表示
        使用法: status
        """
        Display.print_header("水族館ステータス", "🏛️")
        
        # 基本情報
        print(f"{Colors.BRIGHT}水族館名:{Colors.RESET} {self.aquarium.name}")
        print(f"{Colors.BRIGHT}経過日数:{Colors.RESET} {self.aquarium.current_day}日目")
        print(f"{Colors.BRIGHT}資金:{Colors.RESET} ", end="")
        Display.print_money(self.aquarium.money)
        print(f"{Colors.BRIGHT}評判:{Colors.RESET} {self.aquarium.reputation}/100")
        
        Display.print_divider()
        
        # 水槽情報
        print(f"\n{Colors.BRIGHT}🏠 水槽情報{Colors.RESET}")
        print(f"  水槽数: {len(self.aquarium.tanks)}個")
        
        total_fish = sum(len(tank.fish_list) for tank in self.aquarium.tanks)
        print(f"  総魚数: {total_fish}匹")
        
        # 各水槽の簡易表示
        for i, tank in enumerate(self.aquarium.tanks, 1):
            wq = tank.water_quality
            fish_count = len(tank.fish_list)
            clean = int(wq['cleanliness'])
            
            # 状態に応じた色
            if clean >= 80:
                clean_color = Colors.GREEN
            elif clean >= 50:
                clean_color = Colors.YELLOW
            else:
                clean_color = Colors.RED
            
            print(f"  水槽{i}: {fish_count}匹 | 清潔度: {clean_color}{clean}%{Colors.RESET}")
        
        Display.print_divider()
        
        # 今日の収支予測
        expected_income = self.game_engine._calculate_daily_income()
        expected_expenses = self.game_engine._calculate_daily_expenses()
        net_income = expected_income - expected_expenses
        
        print(f"\n{Colors.BRIGHT}💰 今日の収支予測{Colors.RESET}")
        print(f"  予想収入: ", end="")
        Display.print_money(expected_income)
        print(f"  予想支出: ", end="")
        Display.print_money(-expected_expenses)
        print(f"  純利益: ", end="")
        Display.print_money(net_income)
        
        print()
    
    def do_list(self, arg):
        """
        詳細なリストを表示
        使用法: 
            list fish   - すべての魚のリスト
            list tanks  - すべての水槽のリスト
            list all    - すべての詳細情報
        """
        if not arg or arg == "all":
            self._list_all()
        elif arg == "fish":
            self._list_fish()
        elif arg == "tanks":
            self._list_tanks()
        else:
            Display.print_error(f"不明なオプション: {arg}")
            Display.print_info("使用法: list [fish|tanks|all]")
    
    def _list_all(self):
        """すべての情報を表示"""
        self._list_tanks()
        print()
        self._list_fish()
    
    def _list_fish(self):
        """すべての魚のリストを表示"""
        Display.print_section("🐠 魚のリスト")
        
        headers = ["ID", "名前", "種類", "健康", "空腹", "ストレス", "状態"]
        rows = []
        
        for tank_num, tank in enumerate(self.aquarium.tanks, 1):
            for fish in tank.fish_list:
                # 状態の判定
                if fish.is_sick:
                    status = f"{Colors.RED}病気{Colors.RESET}"
                elif fish.hunger > 70:
                    status = f"{Colors.YELLOW}空腹{Colors.RESET}"
                elif fish.stress_level > 70:
                    status = f"{Colors.YELLOW}ストレス{Colors.RESET}"
                else:
                    status = f"{Colors.GREEN}元気{Colors.RESET}"
                
                rows.append([
                    fish.fish_id[:8],
                    fish.name or "-",
                    fish.species,
                    f"{fish.health}%",
                    f"{fish.hunger}%",
                    f"{fish.stress_level}%",
                    status
                ])
        
        if rows:
            Display.print_table(headers, rows, tablefmt="grid")
            print(f"\n総計: {len(rows)}匹")
        else:
            Display.print_warning("魚がいません")
    
    def _list_tanks(self):
        """すべての水槽のリストを表示"""
        Display.print_section("🏠 水槽のリスト")
        
        for i, tank in enumerate(self.aquarium.tanks, 1):
            Display.print_tank_status(tank, i)
    
    def do_feed(self, arg):
        """
        魚に餌を与える
        使用法: 
            feed        - すべての水槽の魚に餌を与える
            feed 1      - 水槽1の魚に餌を与える
            feed all    - すべての水槽の魚に餌を与える
        """
        if not arg or arg == "all":
            # すべての水槽の魚に餌を与える
            total_fed = 0
            for tank in self.aquarium.tanks:
                for fish in tank.fish_list:
                    fish.feed()
                    total_fed += 1
            
            cost = total_fed * 10
            self.aquarium.money -= cost
            
            Display.print_success(f"{total_fed}匹の魚に餌を与えました")
            Display.print_info(f"費用: ¥{cost}")
        else:
            # 特定の水槽の魚に餌を与える
            try:
                tank_num = int(arg)
                if 1 <= tank_num <= len(self.aquarium.tanks):
                    tank = self.aquarium.tanks[tank_num - 1]
                    fed_count = 0
                    for fish in tank.fish_list:
                        fish.feed()
                        fed_count += 1
                    
                    cost = fed_count * 10
                    self.aquarium.money -= cost
                    
                    Display.print_success(f"水槽{tank_num}の{fed_count}匹に餌を与えました")
                    Display.print_info(f"費用: ¥{cost}")
                else:
                    Display.print_error(f"水槽{tank_num}は存在しません")
            except ValueError:
                Display.print_error("無効な水槽番号です")
                Display.print_info("使用法: feed [水槽番号|all]")
    
    def do_clean(self, arg):
        """
        水槽を清掃する
        使用法: clean <水槽番号>
        例: clean 1
        """
        if not arg:
            Display.print_error("水槽番号を指定してください")
            Display.print_info("使用法: clean <水槽番号>")
            return
        
        try:
            tank_num = int(arg)
            if 1 <= tank_num <= len(self.aquarium.tanks):
                tank = self.aquarium.tanks[tank_num - 1]
                old_cleanliness = tank.water_quality['cleanliness']
                tank.clean()
                new_cleanliness = tank.water_quality['cleanliness']
                
                improvement = new_cleanliness - old_cleanliness
                cost = 100
                self.aquarium.money -= cost
                
                Display.print_success(f"水槽{tank_num}を清掃しました")
                Display.print_info(f"清潔度: {old_cleanliness:.0f}% → {new_cleanliness:.0f}% (+{improvement:.0f}%)")
                Display.print_info(f"費用: ¥{cost}")
            else:
                Display.print_error(f"水槽{tank_num}は存在しません")
        except ValueError:
            Display.print_error("無効な水槽番号です")
    
    def do_add_fish(self, arg):
        """
        新しい魚を追加
        使用法: add_fish <種類> [名前]
        例: add_fish goldfish キンちゃん
        """
        parts = arg.split(maxsplit=1)
        if not parts:
            Display.print_error("魚の種類を指定してください")
            Display.print_info("使用法: add_fish <種類> [名前]")
            return
        
        species = parts[0]
        name = parts[1] if len(parts) > 1 else ""
        
        # 魚の価格（種類によって異なる - 今はサンプル）
        price = 1000
        
        if self.aquarium.money < price:
            Display.print_error(f"資金が不足しています（必要: ¥{price}）")
            return
        
        # 空いている水槽を探す
        target_tank = None
        for tank in self.aquarium.tanks:
            if len(tank.fish_list) < tank.capacity // 20:  # 容量に応じた魚数制限
                target_tank = tank
                break
        
        if not target_tank:
            Display.print_error("魚を追加できる水槽がありません")
            Display.print_info("水槽を追加するか、既存の水槽に空きを作ってください")
            return
        
        # 魚を作成して追加
        new_fish = Fish(
            species=species,
            name=name,
            min_temperature=18.0,
            max_temperature=26.0,
            preferred_ph=7.0,
            ph_tolerance=1.0
        )
        
        target_tank.add_fish(new_fish)
        self.aquarium.money -= price
        
        Display.print_success(f"{species}を追加しました")
        if name:
            Display.print_info(f"名前: {name}")
        Display.print_info(f"魚ID: {new_fish.fish_id[:8]}")
        Display.print_info(f"費用: ¥{price}")
    
    def do_add_tank(self, arg):
        """
        新しい水槽を追加
        使用法: add_tank [容量]
        例: add_tank 150
        """
        capacity = 100  # デフォルト容量
        
        if arg:
            try:
                capacity = int(arg)
                if capacity < 50 or capacity > 500:
                    Display.print_error("容量は50〜500Lの範囲で指定してください")
                    return
            except ValueError:
                Display.print_error("無効な容量です")
                return
        
        # 価格は容量に比例
        price = capacity * 50
        
        if self.aquarium.money < price:
            Display.print_error(f"資金が不足しています（必要: ¥{price:,.0f}）")
            return
        
        new_tank = Tank(capacity=capacity)
        self.aquarium.add_tank(new_tank)
        self.aquarium.money -= price
        
        Display.print_success(f"新しい水槽を追加しました")
        Display.print_info(f"容量: {capacity}L")
        Display.print_info(f"費用: ¥{price:,.0f}")
        Display.print_info(f"現在の水槽数: {len(self.aquarium.tanks)}個")

    def do_buy(self, arg):
        """
        アイテムを購入
        使用法: buy <アイテムタイプ> <アイテムID>
        例: buy equipment filter_001
        """
        parts = arg.split()
        if len(parts) != 2:
            Display.print_error("アイテムタイプとIDを指定してください")
            Display.print_info("使用法: buy  ")
            return
        
        item_type, item_id = parts
        
        if item_type == "equipment":
            # サンプル装備データ
            sample_equipment = {
                'filter_001': ('フィルター', 500, 50),
                'heater_001': ('ヒーター', 800, 20),
                'light_001': ('ライト', 300, 15),
            }
            
            if item_id not in sample_equipment:
                Display.print_error(f"装備ID '{item_id}' は見つかりません")
                return
            
            name, price, maintenance = sample_equipment[item_id]
            
            if self.aquarium.money < price:
                Display.print_error(f"資金が不足しています（必要: ¥{price}）")
                return
            
            # 装備を作成して追加
            equipment = Equipment(
                name=name,
                equipment_type="水質管理",
                maintenance_cost=maintenance,
                durability=100
            )
            
            # 最初の水槽に追加（実際は選択できるようにする）
            if self.aquarium.tanks:
                self.aquarium.tanks[0].equipment.append(equipment)
                self.aquarium.money -= price
                
                Display.print_success(f"{name}を購入しました")
                Display.print_info(f"費用: ¥{price}")
                Display.print_info(f"維持費: ¥{maintenance}/日")
            else:
                Display.print_error("装備を設置する水槽がありません")
        
        elif item_type == "decoration":
            Display.print_info("装飾アイテムの購入機能は Phase 4 で実装します")
        else:
            Display.print_error(f"不明なアイテムタイプ: {item_type}")
    
    def do_next_day(self, arg):
        """
        次の日に進む
        使用法: 
            next_day       - 1日進める
            next_day 3     - 3日進める
        """
        days = 1
        if arg:
            try:
                days = int(arg)
                if days < 1 or days > 30:
                    Display.print_error("日数は1〜30の範囲で指定してください")
                    return
            except ValueError:
                Display.print_error("無効な日数です")
                return
        
        Display.print_header(f"{days}日進行中...", "⏰")
        
        for day in range(days):
            # 1日シミュレーション
            result = self.game_engine.simulate_day()
            
            if day == days - 1:  # 最後の日だけ表示
                self._display_day_result(result)
        
        Display.print_success(f"{days}日が経過しました")
        Display.print_info(f"現在: {self.aquarium.current_day}日目")

    def do_skip(self, arg):
        """
        複数日をスキップ（簡易表示）
        使用法: skip <日数>
        例: skip 7
        """
        if not arg:
            Display.print_error("スキップする日数を指定してください")
            Display.print_info("使用法: skip <日数>")
            return
        
        try:
            days = int(arg)
            if days < 1 or days > 90:
                Display.print_error("日数は1〜90の範囲で指定してください")
                return
        except ValueError:
            Display.print_error("無効な日数です")
            return
        
        if not Display.confirm(f"{days}日をスキップしますか？"):
            Display.print_info("キャンセルしました")
            return
        
        Display.print_header(f"{days}日をスキップ中...", "⏩")
        
        total_income = 0
        total_expenses = 0
        total_deaths = 0
        
        for _ in range(days):
            result = self.game_engine.simulate_day()
            total_income += result['income']
            total_expenses += result['expenses']
            if result.get('deaths'):
                total_deaths += len(result['deaths'])
        
        # 結果サマリー
        Display.print_success(f"{days}日が経過しました")
        print(f"\n{Colors.BRIGHT}📊 期間サマリー{Colors.RESET}")
        print(f"  総収入: ", end="")
        Display.print_money(total_income)
        print(f"  総支出: ", end="")
        Display.print_money(-total_expenses)
        print(f"  純利益: ", end="")
        Display.print_money(total_income - total_expenses)
        
        if total_deaths > 0:
            print(f"  {Colors.RED}死亡: {total_deaths}匹{Colors.RESET}")
        
        print(f"\n  現在日: {self.aquarium.current_day}日目")
        print(f"  現在資金: ¥{self.aquarium.money:,.0f}")

    def do_report(self, arg):
        """
        詳細レポートを表示
        使用法: report
        """
        Display.print_header("水族館レポート", "📊")
        
        # 基本統計
        total_fish = sum(len(tank.fish_list) for tank in self.aquarium.tanks)
        avg_health = 0
        if total_fish > 0:
            total_health = sum(
                fish.health 
                for tank in self.aquarium.tanks 
                for fish in tank.fish_list
            )
            avg_health = total_health / total_fish
        
        print(f"{Colors.BRIGHT}📈 統計情報{Colors.RESET}")
        print(f"  経過日数: {self.aquarium.current_day}日")
        print(f"  水槽数: {len(self.aquarium.tanks)}個")
        print(f"  総魚数: {total_fish}匹")
        print(f"  平均健康度: {avg_health:.1f}%")
        print(f"  評判: {self.aquarium.reputation}/100")
        print(f"  資金: ¥{self.aquarium.money:,.0f}")
        
        # 種類別魚数
        species_count = {}
        for tank in self.aquarium.tanks:
            for fish in tank.fish_list:
                species_count[fish.species] = species_count.get(fish.species, 0) + 1
        
        if species_count:
            print(f"\n{Colors.BRIGHT}🐠 種類別魚数{Colors.RESET}")
            for species, count in sorted(species_count.items()):
                print(f"  {species}: {count}匹")
        
        # 収支分析
        expected_income = self.game_engine._calculate_daily_income()
        expected_expenses = self.game_engine._calculate_daily_expenses()
        expense_breakdown = self.game_engine._get_expense_breakdown()
        
        print(f"\n{Colors.BRIGHT}💰 収支分析{Colors.RESET}")
        print(f"  1日の収入予想: ¥{expected_income:,.0f}")
        print(f"  1日の支出予想: ¥{expected_expenses:,.0f}")
        print(f"  1日の純利益予想: ¥{expected_income - expected_expenses:,.0f}")
        
        print(f"\n  支出内訳:")
        for category, amount in expense_breakdown.items():
            print(f"    {category}: ¥{amount:,.0f}")
    
    def do_achievements(self, arg):
        """
        実績を表示
        使用法: achievements
        """
        Display.print_header("実績", "🏆")
        
        achievements = self.game_engine.achievements
        
        if not achievements:
            Display.print_warning("まだ実績を解除していません")
            Display.print_info("ゲームを進めて実績を解除しましょう！")
            return
        
        print(f"{Colors.BRIGHT}解除済み実績: {len(achievements)}個{Colors.RESET}\n")
        
        for achievement in achievements:
            print(f"{Colors.YELLOW}🏆{Colors.RESET} {achievement}")
        
        # 未解除の実績ヒント（サンプル）
        all_achievements = [
            "First Steps - 最初の魚を飼う",
            "Week Survivor - 7日間生存",
            "Month Master - 30日間生存",
            "Collector - 10種類の魚を集める",
            "Wealthy - 資金10,000円達成",
        ]
        
        remaining = [a for a in all_achievements if a not in achievements]
        if remaining:
            print(f"\n{Colors.BRIGHT}未解除の実績（ヒント）:{Colors.RESET}")
            for hint in remaining[:3]:  # 最初の3つだけ表示
                print(f"  ? {hint}")
    
    def do_quit(self, arg):
        """
        ゲームを終了
        使用法: quit
        """
        if Display.confirm("本当に終了しますか？"):
            Display.print_header("ゲーム終了", "👋")
            print("またのご来館をお待ちしています！\n")
            return True
        return False
    
    # エイリアス
    do_exit = do_quit
    do_EOF = do_quit  # Ctrl+Dで終了
