Phase 0: MVP

🎯 Phase 0: MVP（最小限の機能）

まず動くものを作る - シンプルな水族館ゲームの基礎

🎯 このフェーズの目標

Phase 0では、最小限の機能で動作するゲームを作ります。完璧を目指さず、まず「動くもの」を作ることが重要です。

💡 MVPとは？
Minimum Viable Product（最小実行可能製品）の略。必要最小限の機能だけを実装し、早く動くものを作ることで、開発の方向性を確認できます。

達成すべきこと：

  • ✅ 水槽を1つ作成できる
  • ✅ 魚を追加できる（名前と種類のみ）
  • ✅ 魚のリストを表示できる
  • ✅ 簡単なコマンド操作ができる
  • ✅ プログラムを終了できる
⚠️ このフェーズで「やらない」こと：
複雑な水質管理、健康状態、時間経過、セーブ/ロード機能などは後のフェーズで実装します。まずはシンプルに！

2魚クラスの実装（fish.py）

最もシンプルなクラスから始めます。魚は「名前」と「種類」だけを持ちます。

💻 完全なコード（fish.py）：

"""
魚クラス - Phase 0 MVP版
最小限の機能のみ実装
"""

class Fish:
    """
    水族館の魚を表すクラス
    
    Phase 0では名前と種類のみを管理します
    """
    
    def __init__(self, name: str, species: str):
        """
        魚を初期化
        
        Args:
            name (str): 魚の名前（例: "ニモ"）
            species (str): 魚の種類（例: "金魚"）
        """
        self.name = name
        self.species = species
    
    def __str__(self) -> str:
        """
        魚の情報を文字列で返す
        
        Returns:
            str: 魚の情報（例: "ニモ (金魚)"）
        """
        return f"{self.name} ({self.species})"
    
    def __repr__(self) -> str:
        """
        魚のデバッグ用文字列を返す
        
        Returns:
            str: デバッグ情報
        """
        return f"Fish(name='{self.name}', species='{self.species}')"


# ===== テストコード =====
# このファイルを直接実行したときのみ動作
if __name__ == "__main__":
    print("🐠 魚クラスのテスト開始")
    print("-" * 40)
    
    # テスト1: 魚の作成
    fish1 = Fish("ニモ", "クマノミ")
    print(f"✅ テスト1: {fish1}")
    
    # テスト2: 別の魚を作成
    fish2 = Fish("金太郎", "金魚")
    print(f"✅ テスト2: {fish2}")
    
    # テスト3: 魚の名前にアクセス
    print(f"✅ テスト3: {fish1.name}の種類は{fish1.species}です")
    
    print("-" * 40)
    print("🎉 すべてのテスト完了！")

🧪 テスト実行：

📝 コードの解説：
  • __init__: クラスの初期化メソッド（コンストラクタ）
  • __str__: print()で呼ばれる文字列表現
  • __repr__: デバッグ用の詳細な文字列表現
  • if __name__ == "__main__": このファイルを直接実行したときのみ動作

3水槽クラスの実装（tank.py）

魚を入れる水槽を作ります。水槽は複数の魚を管理できます。

💻 完全なコード（tank.py）：

"""
水槽クラス - Phase 0 MVP版
魚のリストを管理するシンプルな実装
"""

from fish import Fish
from typing import List, Optional


class Tank:
    """
    水族館の水槽を表すクラス
    
    Phase 0では魚のリスト管理のみ実装
    """
    
    def __init__(self, name: str = "メイン水槽"):
        """
        水槽を初期化
        
        Args:
            name (str): 水槽の名前
        """
        self.name = name
        self.fish_list: List[Fish] = []
    
    def add_fish(self, fish: Fish) -> bool:
        """
        水槽に魚を追加
        
        Args:
            fish (Fish): 追加する魚
            
        Returns:
            bool: 追加成功ならTrue
        """
        self.fish_list.append(fish)
        return True
    
    def remove_fish(self, fish_name: str) -> bool:
        """
        水槽から魚を削除
        
        Args:
            fish_name (str): 削除する魚の名前
            
        Returns:
            bool: 削除成功ならTrue、魚が見つからなければFalse
        """
        for fish in self.fish_list:
            if fish.name == fish_name:
                self.fish_list.remove(fish)
                return True
        return False
    
    def find_fish(self, fish_name: str) -> Optional[Fish]:
        """
        名前で魚を検索
        
        Args:
            fish_name (str): 検索する魚の名前
            
        Returns:
            Optional[Fish]: 見つかった魚、見つからなければNone
        """
        for fish in self.fish_list:
            if fish.name == fish_name:
                return fish
        return None
    
    def get_fish_count(self) -> int:
        """
        水槽内の魚の数を返す
        
        Returns:
            int: 魚の数
        """
        return len(self.fish_list)
    
    def list_all_fish(self) -> List[Fish]:
        """
        水槽内のすべての魚のリストを返す
        
        Returns:
            List[Fish]: 魚のリスト
        """
        return self.fish_list.copy()
    
    def __str__(self) -> str:
        """
        水槽の情報を文字列で返す
        
        Returns:
            str: 水槽の情報
        """
        fish_count = self.get_fish_count()
        return f"{self.name} (魚: {fish_count}匹)"


# ===== テストコード =====
if __name__ == "__main__":
    print("🏊 水槽クラスのテスト開始")
    print("-" * 40)
    
    # テスト1: 水槽の作成
    tank = Tank("テスト水槽")
    print(f"✅ テスト1: {tank}")
    
    # テスト2: 魚を追加
    fish1 = Fish("ニモ", "クマノミ")
    tank.add_fish(fish1)
    print(f"✅ テスト2: 魚を追加 → {tank}")
    
    # テスト3: 複数の魚を追加
    tank.add_fish(Fish("ドリー", "ナンヨウハギ"))
    tank.add_fish(Fish("金太郎", "金魚"))
    print(f"✅ テスト3: 魚を追加 → {tank}")
    
    # テスト4: 魚のリストを表示
    print("✅ テスト4: 魚リスト:")
    for fish in tank.list_all_fish():
        print(f"   - {fish}")
    
    # テスト5: 魚を検索
    found_fish = tank.find_fish("ニモ")
    if found_fish:
        print(f"✅ テスト5: 魚を検索 → 見つかりました: {found_fish}")
    
    # テスト6: 魚を削除
    tank.remove_fish("ドリー")
    print(f"✅ テスト6: 魚を削除 → {tank}")
    
    print("-" * 40)
    print("🎉 すべてのテスト完了！")

🧪 テスト実行：

# tank.pyを実行してテスト
python tank.py
📝 重要なポイント：
  • List[Fish]: 型ヒント - Fishオブジェクトのリストであることを明示
  • Optional[Fish]: 魚が見つかるか、Noneを返す可能性がある
  • .copy(): リストのコピーを返すことで、元のリストを保護

4メインプログラムの実装（main.py）

ユーザーがコマンドを入力してゲームを操作できるようにします。

💻 完全なコード（main.py）：

"""
バーチャル水族館ゲーム - Phase 0 MVP版
メインプログラム
"""

from tank import Tank
from fish import Fish


def print_header():
    """ゲームのヘッダーを表示"""
    print("\n" + "=" * 50)
    print("🐠 バーチャル水族館ゲーム - Phase 0 MVP")
    print("=" * 50 + "\n")


def print_menu():
    """コマンドメニューを表示"""
    print("\n📋 利用可能なコマンド:")
    print("  1. add    - 魚を追加")
    print("  2. list   - 魚のリストを表示")
    print("  3. remove - 魚を削除")
    print("  4. find   - 魚を検索")
    print("  5. help   - ヘルプを表示")
    print("  6. quit   - ゲームを終了")
    print()


def cmd_add_fish(tank: Tank):
    """魚を追加するコマンド"""
    print("\n🐟 魚を追加します")
    name = input("魚の名前を入力: ").strip()
    
    if not name:
        print("❌ 名前を入力してください")
        return
    
    # 同じ名前の魚がいないかチェック
    if tank.find_fish(name):
        print(f"❌ '{name}'という名前の魚はすでにいます")
        return
    
    species = input("魚の種類を入力（例: 金魚、クマノミ）: ").strip()
    
    if not species:
        species = "不明"
    
    fish = Fish(name, species)
    tank.add_fish(fish)
    print(f"✅ {fish}を追加しました！")


def cmd_list_fish(tank: Tank):
    """魚のリストを表示するコマンド"""
    print("\n🐠 魚のリスト")
    print("-" * 40)
    
    fish_list = tank.list_all_fish()
    
    if not fish_list:
        print("水槽に魚はいません")
    else:
        for i, fish in enumerate(fish_list, 1):
            print(f"{i}. {fish}")
    
    print(f"\n合計: {tank.get_fish_count()}匹")
    print("-" * 40)


def cmd_remove_fish(tank: Tank):
    """魚を削除するコマンド"""
    print("\n🗑️ 魚を削除します")
    name = input("削除する魚の名前を入力: ").strip()
    
    if not name:
        print("❌ 名前を入力してください")
        return
    
    if tank.remove_fish(name):
        print(f"✅ {name}を削除しました")
    else:
        print(f"❌ {name}という名前の魚は見つかりませんでした")


def cmd_find_fish(tank: Tank):
    """魚を検索するコマンド"""
    print("\n🔍 魚を検索します")
    name = input("検索する魚の名前を入力: ").strip()
    
    if not name:
        print("❌ 名前を入力してください")
        return
    
    fish = tank.find_fish(name)
    
    if fish:
        print(f"✅ 見つかりました: {fish}")
    else:
        print(f"❌ {name}という名前の魚は見つかりませんでした")


def main():
    """メイン処理"""
    print_header()
    print("🎮 ゲームを開始します！")
    print("'help'と入力するとコマンド一覧が表示されます")
    
    # 水槽を作成
    tank = Tank("メイン水槽")
    
    # テスト用の魚を追加（オプション）
    tank.add_fish(Fish("サンプル太郎", "金魚"))
    print(f"\n水槽を作成しました: {tank}")
    
    # コマンドループ
    while True:
        command = input("\n🎮 コマンド> ").strip().lower()
        
        if command == "quit" or command == "q":
            print("\n👋 ゲームを終了します。ありがとうございました！")
            break
        
        elif command == "add" or command == "a":
            cmd_add_fish(tank)
        
        elif command == "list" or command == "l":
            cmd_list_fish(tank)
        
        elif command == "remove" or command == "r":
            cmd_remove_fish(tank)
        
        elif command == "find" or command == "f":
            cmd_find_fish(tank)
        
        elif command == "help" or command == "h":
            print_menu()
        
        elif command == "":
            # 何も入力されなかった場合はスキップ
            continue
        
        else:
            print(f"❌ 不明なコマンド: '{command}'")
            print("'help'と入力してコマンド一覧を確認してください")


if __name__ == "__main__":
    try:
        main()
    except KeyboardInterrupt:
        print("\n\n👋 ゲームを中断しました")
    except Exception as e:
        print(f"\n❌ エラーが発生しました: {e}")

🐛 よくある問題と解決方法

問題1: ModuleNotFoundError: No module named 'fish'

原因： fish.pyやtank.pyが同じディレクトリにない
解決方法：
  • main.py、fish.py、tank.pyが同じフォルダにあるか確認
  • 正しいディレクトリで実行しているか確認: ls -la

問題2: KeyboardInterrupt が正しく動作しない

原因： Ctrl+Cで中断したときの処理
解決方法： これは正常な動作です。quitコマンドで終了するか、Ctrl+Cで強制終了できます

問題3: 日本語が文字化けする

原因： ターミナルの文字コード設定
解決方法：
  • ファイルをUTF-8で保存
  • ターミナルの文字コードをUTF-8に設定
  • Windowsの場合: chcp 65001 を実行
