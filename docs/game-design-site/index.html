📑 目次

🎮 ゲーム概要

バーチャル水族館ゲームは、プレイヤーが自分だけの水族館を管理・運営するCUI（キャラクターユーザーインターフェース）ベースのシミュレーションゲームです。魚の飼育、水槽の管理、装飾の配置を通じて、美しく健全な水族館を作り上げます。

ゲームの目標

  • 魚の健康維持：適切な餌やり、水質管理により魚を健康に保つ
  • 水族館の発展：来客数と評価を高め、収益を増やす
  • コレクション完成：様々な種類の魚や装飾を集める
  • 美観の追求：デザイン性の高い水槽レイアウトを実現

スコアリング基準

評価項目 配点 詳細
魚の健康度 40% 全魚の平均健康値（0-100）
水質スコア 25% pH、温度、清潔度の総合評価
美観スコア 20% 装飾配置とバランス
来客満足度 15% 訪問者の評価平均

状態管理

ゲームは内部で以下の状態を管理します：

  • ゲーム時間：日数、季節の変化
  • 資金：収入（入場料）、支出（餌、装備、メンテナンス）
  • 水槽状態：各水槽の水質パラメータ、温度、容量
  • 魚の状態：健康度、空腹度、ストレスレベル、成長段階
  • イベント：来客、魚の繁殖、病気発生、特別イベント

# models.py の一部
from dataclasses import dataclass
from typing import List, Dict
import uuid

@dataclass
class Fish:
    """魚クラス - 個別の魚の状態を管理"""
    fish_id: str
    species: str
    name: str
    health: int = 100
    hunger: int = 0
    age: int = 0
    size: float = 5.0
    stress_level: int = 0
    price: int = 100
    
    def __post_init__(self):
        if not self.fish_id:
            self.fish_id = str(uuid.uuid4())
    
    def feed(self, amount: int = 20):
        """餌を与える"""
        self.hunger = max(0, self.hunger - amount)
        self.health = min(100, self.health + amount // 4)
    
    def update_status(self, environment: Dict):
        """環境に基づいて状態を更新"""
        # 空腹度増加
        self.hunger = min(100, self.hunger + 5)
        
        # 健康度の変化
        if self.hunger > 80:
            self.health -= 10
        
        # 水質による影響
        water_quality = environment.get('cleanliness', 50)
        if water_quality < 30:
            self.health -= 5
            self.stress_level = min(100, self.stress_level + 10)
        
        # 年齢増加
        self.age += 1
        self.grow()
    
    def grow(self):
        """成長処理"""
        if self.age % 30 == 0:  # 30日ごとに成長
            self.size *= 1.1
    
    def calculate_happiness(self) -> int:
        """幸福度を計算"""
        happiness = 50
        happiness += (100 - self.hunger) // 5
        happiness += (100 - self.stress_level) // 5
        happiness += self.health // 5
        return min(100, max(0, happiness))

class Tank:
    """水槽クラス"""
    def __init__(self, tank_id: str, capacity: int):
        self.tank_id = tank_id
        self.capacity = capacity
        self.water_quality = {
            'ph': 7.0,
            'temperature': 25.0,
            'cleanliness': 100,
            'oxygen': 100
        }
        self.fish_list: List[Fish] = []
        self.equipment: List[Equipment] = []
        self.decorations: List[Decoration] = []
    
    def add_fish(self, fish: Fish) -> bool:
        """魚を追加"""
        current_load = sum(f.size for f in self.fish_list)
        if current_load + fish.size <= self.capacity:
            self.fish_list.append(fish)
            return True
        return False
    
    def update_water_quality(self, days: int = 1):
        """水質を経時変化させる"""
        # 清潔度低下
        self.water_quality['cleanliness'] -= days * 5
        self.water_quality['cleanliness'] = max(0, self.water_quality['cleanliness'])
        
        # 酸素濃度変化
        fish_count = len(self.fish_list)
        oxygen_consumption = fish_count * 2
        self.water_quality['oxygen'] -= oxygen_consumption
        self.water_quality['oxygen'] = max(20, self.water_quality['oxygen'])
    
    def clean_tank(self):
        """水槽を清掃"""
        self.water_quality['cleanliness'] = 100
        self.water_quality['oxygen'] = 100

# logic.py の一部
import random
from models import Aquarium, Tank, Fish

class GameEngine:
    """ゲームエンジン - コアロジック"""
    
    def __init__(self, aquarium: Aquarium):
        self.aquarium = aquarium
        self.event_log = []
    
    def simulate_day(self) -> Dict:
        """1日のシミュレーション"""
        events = []
        
        # 各水槽の処理
        for tank in self.aquarium.tanks:
            # 水質更新
            tank.update_water_quality()
            
            # 各魚の状態更新
            for fish in tank.fish_list:
                environment = tank.water_quality.copy()
                fish.update_status(environment)
                
                # 魚の死亡判定
                if fish.health <= 0:
                    events.append({
                        'type': 'fish_died',
                        'fish': fish.name,
                        'tank': tank.tank_id
                    })
                    tank.fish_list.remove(fish)
            
            # 装備の効果適用
            for equipment in tank.equipment:
                equipment.apply_effect(tank)
                equipment.degrade(1)
        
        # 来客処理
        visitors = self.calculate_visitors()
        income = visitors * 10  # 1人あたり10円
        self.aquarium.funds += income
        
        events.append({
            'type': 'daily_income',
            'visitors': visitors,
            'income': income
        })
        
        # 日数カウント
        self.aquarium.current_day += 1
        
        # イベント発生判定
        if random.random() < 0.1:  # 10%の確率
            event = self.generate_random_event()
            events.append(event)
        
        self.event_log.extend(events)
        return {
            'day': self.aquarium.current_day,
            'events': events,
            'funds': self.aquarium.funds
        }
    
    def calculate_visitors(self) -> int:
        """来客数を計算"""
        base_visitors = 50
        
        # 評判によるボーナス
        reputation_bonus = self.aquarium.reputation * 2
        
        # 魚の総数によるボーナス
        total_fish = sum(len(tank.fish_list) for tank in self.aquarium.tanks)
        fish_bonus = total_fish * 5
        
        # 美観スコアによるボーナス
        beauty_score = sum(tank.calculate_beauty_score() for tank in self.aquarium.tanks)
        beauty_bonus = beauty_score // 10
        
        total = base_visitors + reputation_bonus + fish_bonus + beauty_bonus
        return max(0, total + random.randint(-20, 20))
    
    def feed_all_fish(self, tank_id: str):
        """特定水槽の全魚に餌やり"""
        tank = self.get_tank_by_id(tank_id)
        if tank:
            for fish in tank.fish_list:
                fish.feed()
            return True
        return False
    
    def update_water_quality_all(self):
        """全水槽の水質を更新"""
        for tank in self.aquarium.tanks:
            tank.update_water_quality()
    
    def generate_random_event(self) -> Dict:
        """ランダムイベント生成"""
        events = [
            {'type': 'donation', 'amount': 1000, 'message': '匿名の寄付がありました！'},
            {'type': 'inspection', 'message': '保健所の検査がありました'},
            {'type': 'media', 'reputation': 10, 'message': 'メディアに取り上げられました！'},
        ]
        return random.choice(events)

# ui.py の一部
import cmd
from logic import GameEngine
from models import Aquarium, Tank, Fish

class AquariumCUI(cmd.Cmd):
    """水族館ゲームのCUI"""
    
    intro = '''
╔═══════════════════════════════════════╗
║   🐠 バーチャル水族館ゲーム 🐠     ║
║   'help'でコマンド一覧を表示       ║
╚═══════════════════════════════════════╝
'''
    prompt = '🐠 > '
    
    def __init__(self, aquarium: Aquarium):
        super().__init__()
        self.aquarium = aquarium
        self.engine = GameEngine(aquarium)
    
    def do_status(self, arg):
        """水族館の状態を表示
        使用法: status [tank_id]"""
        if arg:
            self.show_tank_status(arg)
        else:
            self.show_aquarium_status()
    
    def show_aquarium_status(self):
        """水族館全体の状態表示"""
        print("\n" + "="*50)
        print(f"🏛️  {self.aquarium.name}")
        print(f"💰 資金: ¥{self.aquarium.funds:,}")
        print(f"📅 経過日数: {self.aquarium.current_day}日")
        print(f"⭐ 評判: {self.aquarium.reputation}/100")
        print(f"🚪 累計来客: {self.aquarium.visitors_count:,}人")
        print("="*50)
        
        print(f"\n🏺 水槽一覧 ({len(self.aquarium.tanks)}個)")
        for i, tank in enumerate(self.aquarium.tanks, 1):
            fish_count = len(tank.fish_list)
            quality = tank.water_quality['cleanliness']
            status_icon = '✅' if quality > 70 else '⚠️' if quality > 40 else '❌'
            print(f"  {i}. {tank.tank_id} - {fish_count}匹 {status_icon}")
    
    def do_feed(self, arg):
        """魚に餌を与える
        使用法: feed  [fish_name]"""
        args = arg.split()
        if not args:
            print("❌ エラー: 水槽IDを指定してください")
            return
        
        tank_id = args[0]
        if self.engine.feed_all_fish(tank_id):
            print(f"✅ {tank_id}の全魚に餌を与えました")
        else:
            print(f"❌ エラー: 水槽'{tank_id}'が見つかりません")
    
    def do_next_day(self, arg):
        """時間を進める
        使用法: next_day [日数]"""
        days = int(arg) if arg and arg.isdigit() else 1
        
        for _ in range(days):
            result = self.engine.simulate_day()
            
            # イベント表示
            for event in result['events']:
                self.display_event(event)
        
        print(f"\n⏭️  {days}日が経過しました")
        print(f"💰 現在の資金: ¥{result['funds']:,}")
    
    def display_event(self, event):
        """イベントを表示"""
        event_type = event.get('type')
        if event_type == 'fish_died':
            print(f"💀 {event['fish']}が死亡しました ({event['tank']})")
        elif event_type == 'daily_income':
            print(f"🚪 本日の来客: {event['visitors']}人 (+¥{event['income']})")
        elif event_type == 'donation':
            print(f"🎁 {event['message']} (+¥{event['amount']})")
    
    def do_quit(self, arg):
        """ゲームを終了"""
        print("\n👋 またのご来館をお待ちしております！")
        return True
    
    def do_EOF(self, arg):
        """Ctrl+Dで終了"""
        return self.do_quit(arg)

👨‍💻 開発ガイダンス

  1. Phase 1: 基本構造の実装
    • models.pyにクラス定義を作成
    • 基本的な属性とメソッドを実装
    • ユニットテストを作成
  2. Phase 2: ゲームロジックの実装
    • logic.pyにGameEngineクラスを実装
    • シミュレーション処理を実装
    • 統合テストを作成
  3. Phase 3: UIの実装
    • ui.pyにCUIを実装
    • コマンド処理を実装
    • 表示フォーマットを整える
  4. Phase 4: データファイルの作成
    • 魚種、装備、装飾のJSONデータを作成
    • バランス調整
  5. Phase 5: 追加機能の実装
    • セーブ/ロード機能
    • イベントシステム
    • 実績システム
